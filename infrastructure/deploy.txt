Step 2:
Migrations
1. Create migration scripts
  - Enums
  PGPASSWORD=$DB_PASS PGSSLMODE=require PGGSSENCMODE=disable psql \
  -h analytics-staging.c0q4keadjybw.us-east-2.rds.amazonaws.com \
  -p 5432 -U analytics_staging -d analytics -At \
  -c "SELECT 'CREATE TYPE ' || t.typname || ' AS ENUM (' || string_agg(quote_literal(e.enumlabel), ', ' ORDER BY e.enumsortorder) || ');' FROM pg_type t JOIN pg_enum e ON t.oid = e.enumtypid WHERE t.typnamespace = 'public'::regnamespace GROUP BY t.typname;" \
  > /Users/dashrath/Documents/GA4GH/Analytics/analytics-dashboard/liquibase/schema.sql
  
  - Functions
  PGPASSWORD=$DB_PASS PGSSLMODE=require PGGSSENCMODE=disable psql \
  -h analytics-staging.c0q4keadjybw.us-east-2.rds.amazonaws.com \
  -p 5432 -U analytics_staging -d analytics -At \
  -c "SELECT pg_get_functiondef(p.oid) || ';' FROM pg_proc p JOIN pg_namespace n ON p.pronamespace = n.oid WHERE n.nspname = 'public';" \
  >> /Users/dashrath/Documents/GA4GH/Analytics/analytics-dashboard/liquibase/schema.sql
  
  - Triggers
  PGPASSWORD=$DB_PASS PGSSLMODE=require PGGSSENCMODE=disable psql \
  -h analytics-staging.c0q4keadjybw.us-east-2.rds.amazonaws.com \
  -p 5432 -U analytics_staging -d analytics -At \
  -c "SELECT pg_get_triggerdef(oid) || ';' FROM pg_trigger WHERE NOT tgisinternal;" \
  >> /Users/dashrath/Documents/GA4GH/Analytics/analytics-dashboard/liquibase/schema.sql

  - Changelog
  liquibase generate-changelog \
  --driver=org.postgresql.Driver \

2. Deploy migration stack
aws cloudformation deploy \
  --template-file infrastructure/migrations.yml \
  --stack-name analytics-dashboard-staging-migrations \
  --region us-east-2 \
  --parameter-overrides \
    Environment=staging \
    LiquibaseImage=282236276875.dkr.ecr.us-east-2.amazonaws.com/analytics-dashboard-liquibase:latest \
    LiquibaseJdbcUrl=jdbc:postgresql://analytics-dashboard-staging-base-database-verhcap8spla.c0q4keadjybw.us-east-2.rds.amazonaws.com:5432/analytics_dashboard_staging \
    LiquibaseDbUsername=analytics_dashboard_writer \
    DBSecretArn=arn:aws:secretsmanager:us-east-2:282236276875:secret:analytics-dashboard-staging-DBPassword-hRcuSI \
    ExecutionRole=arn:aws:iam::282236276875:role/ECSFargateExecutionRole-TT \
    Region=us-east-2

3. Run migration stack
  aws ecs run-task \
  --cluster analytics-dashboard-staging-cluster \
  --task-definition analytics-dashboard-staging-liquibase \
  --launch-type FARGATE \
  --network-configuration "awsvpcConfiguration={subnets=[subnet-0f26c31e6c9578717,subnet-00663ce89eb47404f,subnet-08b697026fc4502e5],securityGroups=[sg-0f473b33e2fa1c95a],assignPublicIp=ENABLED}" \
  --region us-east-2

  4. Check status
  aws ecs describe-tasks \
  --cluster analytics-dashboard-staging-cluster \
  --tasks <TASK_ARN> \
  --region us-east-2 \
  --query "tasks[0].containers[0].{Status:lastStatus,ExitCode:exitCode,Reason:reason}"

  5. Check logs
  aws logs tail /analytics-dashboard/staging/backend --follow --region us-east-2
