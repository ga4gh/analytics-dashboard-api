=== Step 3: Data Migration (from local production DB) ===

# 3a. Dump local production data (custom format, data-only, excludes Liquibase tables)
pg_dump \
  --data-only \
  --no-owner \
  --no-privileges \
  --exclude-table=databasechangelog \
  --exclude-table=databasechangeloglock \
  -Fc \
  -h localhost -p 5432 \
  -U <local-prod-username> \
  -d <local-prod-dbname> \
  -f /tmp/prod_data_dump.dump

# 3b. Get production DB password
PROD_DBPASS=$(aws secretsmanager get-secret-value \
  --secret-id analytics-dashboard-production-DBPassword \
  --version-stage AWSCURRENT --region us-east-2 \
  --query 'SecretString' --output text)

# 3c. Restore to production RDS
PGPASSWORD=$PROD_DBPASS PGSSLMODE=require PGGSSENCMODE=disable pg_restore \
  --data-only \
  --no-owner \
  --no-privileges \
  --single-transaction \
  -h <prod-rds-endpoint> -p 5432 \
  -U <prod-username> \
  -d <prod-dbname> \
  /tmp/prod_data_dump.dump

# 3d. Verify row counts
PGPASSWORD=$PROD_DBPASS PGSSLMODE=require PGGSSENCMODE=disable psql \
  -h <prod-rds-endpoint> -p 5432 -U <prod-username> -d <prod-dbname> \
  -c "SELECT relname, n_live_tup FROM pg_stat_user_tables WHERE n_live_tup > 0 ORDER BY n_live_tup DESC;"
